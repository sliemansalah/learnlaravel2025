```javascript
function gradeFillBlank(question, userAnswer) {
    const answer = userAnswer.answer.trim();
    const normalized = question.caseSensitive
        ? answer
        : answer.toLowerCase();

    let isCorrect = false;
    question.correctAnswers.forEach(correct => {
        const normalizedCorrect = question.caseSensitive
            ? correct
            : correct.toLowerCase();
        if (normalized === normalizedCorrect) {
            isCorrect = true;
        }
    });

    return {
        score: isCorrect ? 1 : 0,
        isCorrect
    };
}
```

4. **أوامر (Commands)**
```javascript
function gradeCommand(question, userAnswer) {
    const command = userAnswer.answer.trim();
    const normalized = question.caseSensitive
        ? command
        : command.toLowerCase();

    let isCorrect = false;
    question.acceptedAnswers.forEach(accepted => {
        const normalizedAccepted = question.caseSensitive
            ? accepted
            : accepted.toLowerCase();
        if (normalized === normalizedAccepted) {
            isCorrect = true;
        }
    });

    return {
        score: isCorrect ? 1 : 0,
        isCorrect
    };
}
```
